1. BASIC SOQL
============================================================

SELECT Id, Name, Email
FROM Account


SELECT Id, FirstName, LastName, Email
FROM Contact


SELECT Id, FirstName, LastName, Company, Status
FROM Lead


SELECT Id, Name, Amount, StageName
FROM Opportunity


2. WHERE CLAUSE
============================================================

SELECT Id, Name
FROM Account
WHERE Industry = 'Technology'


SELECT Id, Name
FROM Account
WHERE Industry != 'Technology'


SELECT Id, Name
FROM Account
WHERE AnnualRevenue > 1000000


SELECT Id, Name
FROM Account
WHERE AnnualRevenue < 1000000


SELECT Id, Name
FROM Account
WHERE NumberOfEmployees >= 500


SELECT Id, Name
FROM Account
WHERE NumberOfEmployees <= 500


SELECT Id, Name
FROM Account
WHERE Phone = NULL


SELECT Id, Name
FROM Account
WHERE Phone != NULL


3. AND / OR / NOT
============================================================

SELECT Id, Name
FROM Account
WHERE Industry = 'Technology'
AND Type = 'Customer'


SELECT Id, Name

SELECT Id, Name
FROM Account

SELECT Id, Name
FROM Account

AND (Type = 'Customer' OR Type = 'Prospect')

7. ORDER BY
============================================================

FROM Account
ORDER BY Name ASC


SELECT Id, Name
FROM Account

FROM Account

============================================================

11. GROUP BY
============================================================

SELECT Industry, COUNT(Id)
FROM Account
GROUP BY Industry


SELECT Type, COUNT(Id)
FROM Account
GROUP BY Type


SELECT StageName, SUM(Amount)
FROM Opportunity
GROUP BY StageName


SELECT Industry, AVG(AnnualRevenue)
FROM Account
GROUP BY Industry


SELECT Industry, Type, COUNT(Id)
FROM Account
GROUP BY Industry, Type


12. HAVING
============================================================

SELECT Industry, COUNT(Id)
FROM Account
GROUP BY Industry
HAVING COUNT(Id) > 10


SELECT AccountId, SUM(Amount)
FROM Opportunity
GROUP BY AccountId
HAVING SUM(Amount) > 1000000


13. CHILD-TO-PARENT RELATIONSHIP
============================================================

       Account.Name,
       Account.Industry
FROM Contact


SELECT Id, FirstName, LastName,
       Account.Name,
       Account.Owner.Name
FROM Contact


14. PARENT-TO-CHILD RELATIONSHIP
============================================================

SELECT Id, Name,
       (SELECT Id, FirstName, LastName, Email
        FROM Contacts)
FROM Account


SELECT Id, Name,
       (SELECT Id, FirstName, LastName
        FROM Contacts
        WHERE Email != NULL)
FROM Account


SELECT Id, Name,
       (SELECT Id, FirstName, LastName
        FROM Contacts
        ORDER BY LastName ASC)
FROM Account


SELECT Id, Name,
       (SELECT Id, FirstName, LastName
        FROM Contacts
        LIMIT 5)
FROM Account

18. TYPEOF - END
============================================================

SELECT Id, Subject,
       TYPEOF Owner
           WHEN User THEN Name, Email
           WHEN Group THEN Name
       END
FROM Task


SELECT Id, Subject,
       TYPEOF What
           WHEN Account THEN Name, Industry
           WHEN Opportunity THEN Name, Amount
       END
FROM Task

20. USING SCOPE
============================================================

SELECT Id, Name
FROM Account
USING SCOPE Mine


SELECT Id, FirstName, LastName
FROM Contact
USING SCOPE Mine
                                
	Scope						| Meaning	Simple 													| example
	----------------------------|-------------------------------------------------------------------| ----------------------------------------------
	Mine						| Records owned by you												| Contacts where you are the owner
	MyGroups					| Records owned by your groups/queues								| Records owned by a queue/group you're part of
	MyTerritory					| Records belonging to your territory								| Accounts assigned to your territory
	MyTerritoryAndSubordinates	| Records in your territory + child/subordinate territories			| Your territory + territories underneath it
	Team						| Records where you are a team member								| Accounts where you're on the Account Team
	Delegated					| Records delegated to you											| Records another user has delegated to you

47. COMPLETE SOQL QUERY
============================================================

SELECT AccountId,
       COUNT(Id) OpportunityCount,
       SUM(Amount) TotalAmount,
       AVG(Amount) AverageAmount
FROM Opportunity
WHERE IsClosed = false
AND CloseDate = THIS_YEAR
GROUP BY AccountId
HAVING COUNT(Id) > 2
AND SUM(Amount) > 100000
ORDER BY SUM(Amount) DESC
LIMIT 10
